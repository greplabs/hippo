1//! Result reranking.
2//!
3//! This is the second-stage precision pass of the "elite retrieval" pipeline.
4//! It runs AFTER Reciprocal Rank Fusion (RRF) merges the semantic and keyword
5//! arms, and BEFORE the final truncation to the caller's `limit`:
6//!
7//! ```text
8//! semantic (Qdrant, chunk roll-up) ⨁ FTS5 BM25
9//!     → reciprocal_rank_fusion            (first-stage recall / coarse order)
10//!     → rerank_stage (this module)        (second-stage precision re-order)
11//!     → truncate(limit)                   (final results)
12//! ```
13//!
14//! First-stage fusion maximises *recall* cheaply: RRF is rank-based and
15//! scale-free, so it is good at getting the right documents into the candidate
16//! set, but its ordering ignores strong precision signals (an exact title hit
17//! is worth more than being ranked 3rd in both arms). The reranker re-scores a
18//! bounded window of top candidates using those signals and reorders them.
19//!
20//! Two implementations share the [`Reranker`] trait seam:
21//!
22//! * [`HeuristicReranker`] — the DEFAULT. Zero new dependencies, fully offline
23//!   and deterministic. It re-scores using signals already attached to each
24//!   candidate (fused score, lexical title/filename/tag matches, phrase vs
25//!   scattered-term matches, favorite, recency). This is a real precision win
26//!   over raw RRF order with no extra infrastructure.
27//! * `OnnxReranker` — an OPTIONAL, feature-gated scaffold (see the bottom of
28//!   this file) for a future cross-encoder model. It is NOT wired to a model and
29//!   adds NO dependency to the default build.
30

35/// Reorders a set of already-retrieved candidates to improve precision.
36///
37/// Object-safe by design: no generic methods, no `Self`-by-value returns, so a
38/// `&dyn Reranker` / `Arc<dyn Reranker>` can be stored on the searcher and a
39/// different backend (heuristic today, a cross-encoder later) dropped in without
40/// touching the pipeline. Implementations must be cheap enough to run on every
41/// hybrid query over the top-N candidates.
42pub trait Reranker: Send + Sync {
43    /// Re-order `candidates` for `query`, best-first. Implementations may also
44    /// overwrite each result's `score` with the new rerank score; the caller
45    /// treats the returned order (not the score magnitude) as authoritative.
46    fn rerank(&self, query: &str, candidates: Vec<MemorySearchResult>) -> Vec<MemorySearchResult>;
47
48    /// Short name for logging / telemetry.
49    fn name(&self) -> &'static str {
50        "reranker"
51    }
52}
53
54/// Transparent, documented weights for [`HeuristicReranker`].
55///
56/// The final per-candidate score is a simple weighted sum (see
57/// [`HeuristicReranker::heuristic_score`]):
58///
59/// ```text
60/// score = base   * (fused_score / max_fused_score)   // normalised recall signal, ~[0,1]
61///       + exact      if query == title/filename/stem   // strongest lexical signal
62///       + phrase     if full query is a substring hit   // contiguous match (implied by exact)
63///       + coverage * (matched_terms / total_terms)      // partial / scattered-term match
64///       + tag        if a tag equals a query term/query  // curated label match
65///       + favorite   if the file is starred              // user-intent tie-break
66///       + recency    if modified within 7d / 30d         // freshness tie-break
67/// ```
68///
69/// Rationale for the magnitudes:
70/// * `base = 1.0` keeps the fused recall order as the backbone, but
71/// * `exact = 3.0` and `phrase = 1.5` are large enough to pull a strongly
72///   lexically-matching candidate above one that merely ranked higher in RRF —
73///   the whole point of a precision rerank.
74/// * `coverage = 1.0` rewards matching more query terms, but a *contiguous*
75///   phrase match (exact + phrase) always beats the same terms scattered.
76/// * `favorite` / `recency` are deliberately small: they only decide ties
77///   between otherwise-equivalent candidates.
78#[derive(Debug, Clone, Copy)]
79pub struct RerankWeights {
80    pub base: f32,
81    pub exact: f32,
82    pub phrase: f32,
83    pub coverage: f32,
84    pub tag: f32,
85    pub favorite: f32,
86    pub recent_week: f32,
87    pub recent_month: f32,
88}
89
90impl Default for RerankWeights {
91    fn default() -> Self {
92        Self {
93            base: 1.0,
94            exact: 3.0,
95            phrase: 1.5,
96            coverage: 1.0,
97            tag: 0.8,
98            favorite: 0.3,
99            recent_week: 0.2,
100            recent_month: 0.1,
101        }
102    }
103}

105/// The default, zero-dependency reranker.
106///
107/// Re-scores candidates using signals already present on each
108/// [`MemorySearchResult`] — no model, no network, fully deterministic.
109#[derive(Debug, Clone, Default)]
110pub struct HeuristicReranker {
111    weights: RerankWeights,
112}
113
114impl HeuristicReranker {
115    /// Reranker with the documented default [`RerankWeights`].
116    pub fn new() -> Self {
117        Self::default()
118    }
119
120    /// Reranker with custom weights.
121    pub fn with_weights(weights: RerankWeights) -> Self {
122        Self { weights }
123    }
124
125    /// Weighted heuristic score for a single candidate. `max_base` is the
126    /// largest fused score in the current candidate window (for normalisation),
127    /// `now` is captured once per rerank call for deterministic recency.
128    fn heuristic_score(
129        &self,
130        query_lower: &str,
131        terms: &[&str],
132        c: &MemorySearchResult,
133        max_base: f32,
134        now: chrono::DateTime<Utc>,
135    ) -> f32 {
136        let w = &self.weights;
137
138        // Normalised recall backbone (fused RRF score scaled to ~[0,1]).
139        let mut score = w.base * (c.score / max_base);
140
141        // Favorite + recency are always available as mild tie-breaks, even for
142        // an empty query.
143        if c.memory.is_favorite {
144            score += w.favorite;
145        }
146        let age_days = (now - c.memory.modified_at).num_days();
147        if age_days <= 7 {
148            score += w.recent_week;
149        } else if age_days <= 30 {
150            score += w.recent_month;
151        }
152
153        if query_lower.is_empty() {
154            return score;
155        }
156
157        let title = c
158            .memory
159            .metadata
160            .title
161            .as_deref()
162            .unwrap_or("")
163            .to_lowercase();
164        let filename = c
165            .memory
166            .path
167            .file_name()
168            .map(|f| f.to_string_lossy().to_lowercase())
169            .unwrap_or_default();
170        let stem = c
171            .memory
172            .path
173            .file_stem()
174            .map(|f| f.to_string_lossy().to_lowercase())
175            .unwrap_or_default();
176        let preview = c
177            .memory
178            .metadata
179            .text_preview
180            .as_deref()
181            .unwrap_or("")
182            .to_lowercase();
183        let tags: Vec<String> = c
184            .memory
185            .tags
186            .iter()
187            .map(|t| t.name.to_lowercase())
188            .collect();
189
190        // Exact match: the query IS the title, filename, or filename stem.
191        if title == query_lower || filename == query_lower || stem == query_lower {
192            score += w.exact;
193        }
194
195        // Phrase match: the full query appears as a contiguous substring. Exact
196        // matches also satisfy this (monotonic: exact is strictly best), and a
197        // contiguous phrase always outscores the same terms scattered.
198        if title.contains(query_lower)
199            || filename.contains(query_lower)
200            || preview.contains(query_lower)
201        {
202            score += w.phrase;
203        }
204
205        // Term coverage: fraction of query terms found anywhere in the candidate.
206        if !terms.is_empty() {
207            let haystack = format!("{} {} {} {}", title, filename, preview, tags.join(" "));
208            let hits = terms.iter().filter(|t| haystack.contains(**t)).count();
209            score += w.coverage * (hits as f32 / terms.len() as f32);
210        }
211
212        // Tag match: a curated tag equals the whole query or one of its terms.
213        let tag_match = tags
214            .iter()
215            .any(|tag| tag == query_lower || terms.iter().any(|t| tag == t));
216        if tag_match {
217            score += w.tag;
218        }
219
220        score
221    }
222}

224impl Reranker for HeuristicReranker {
225    fn rerank(&self, query: &str, candidates: Vec<MemorySearchResult>) -> Vec<MemorySearchResult> {
226        if candidates.len() <= 1 {
227            return candidates;
228        }
229
230        let query_lower = query.trim().to_lowercase();
231        let terms: Vec<&str> = query_lower.split_whitespace().collect();
232
233        // Normalise the fused score against the best in this window. Guard a
234        // zero/negative max so division is always well-defined.
235        let max_base = candidates
236            .iter()
237            .map(|c| c.score)
238            .fold(0.0_f32, f32::max)
239            .max(f32::EPSILON);
240
241        let now = Utc::now();
242
243        let mut scored: Vec<(f32, MemorySearchResult)> = candidates
244            .into_iter()
245            .map(|c| {
246                let s = self.heuristic_score(&query_lower, &terms, &c, max_base, now);
247                (s, c)
248            })
249            .collect();
250
251        // Stable sort by descending rerank score: candidates that tie keep their
252        // incoming (fused) relative order, so equal scores never reshuffle.
253        scored.sort_by(|a, b| b.0.partial_cmp(&a.0).unwrap_or(Ordering::Equal));
254
255        scored
256            .into_iter()
257            .map(|(s, mut c)| {
258                c.score = s;
259                c
260            })
261            .collect()
262    }
263
264    fn name(&self) -> &'static str {
265        "heuristic"
266    }
267}

269/// Apply the (gated) rerank stage to a fused, best-first result list.
270///
271/// This is the single entry point the search pipeline calls, kept as a free
272/// function so it is exhaustively testable offline:
273///
274/// * `enabled == false` → identity (returns `results` untouched, i.e. today's
275///   RRF order).
276/// * only the top `rerank_candidates` (by incoming fused order) are reranked;
277///   the tail keeps its RRF order and is appended unchanged. The combined list
278///   is NOT re-sorted, so the head (reranked) always precedes the tail.
279pub fn rerank_stage(
280    reranker: &dyn Reranker,
281    query: &str,
282    mut results: Vec<MemorySearchResult>,
283    enabled: bool,
284    rerank_candidates: usize,
285) -> Vec<MemorySearchResult> {
286    if !enabled || rerank_candidates == 0 || results.len() <= 1 {
287        return results;
288    }
289
290    let cutoff = rerank_candidates.min(results.len());
291    let tail = results.split_off(cutoff);
292    let mut head = reranker.rerank(query, results);
293    head.extend(tail);
294    head
295}
