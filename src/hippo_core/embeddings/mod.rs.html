21/// Embedding dimension for different model types.
22///
23/// IMPORTANT: `TEXT_EMBEDDING_DIM` MUST match the runtime output dim of the
24/// default local embedding model (Ollama `nomic-embed-text` → 768, see
25/// `crate::ollama::NOMIC_EMBED_DIM`). The hash fallback also emits this dim so
26/// that fallback vectors live in the same space as Ollama vectors. A regression
27/// here silently breaks vector storage (dim mismatch on Qdrant upsert).
28pub const TEXT_EMBEDDING_DIM: usize = 768; // nomic-embed-text (default local model)
29pub const IMAGE_EMBEDDING_DIM: usize = 512; // CLIP compatible
30pub const CODE_EMBEDDING_DIM: usize = 768; // CodeBERT compatible / nomic
31/// OpenAI text-embedding-3-small / ada-002 dimension (only used when the OpenAI
32/// backend is the active provider).
33pub const OPENAI_EMBEDDING_DIM: usize = 1536;

40/// A slice of a document produced by [`Embedder::chunk_document`].
41///
42/// `char_start`/`char_end` are BYTE offsets into the original text, so
43/// `&text[char_start..char_end] == text` of the chunk (for ASCII these equal
44/// character offsets).
45#[derive(Debug, Clone, PartialEq, Eq)]
46pub struct TextChunk {
47    pub text: String,
48    pub char_start: usize,
49    pub char_end: usize,
50}

52/// Cache configuration
53const EMBEDDING_CACHE_SIZE: usize = 5000;
54const EMBEDDING_CACHE_TTL_SECS: u64 = 3600; // 1 hour TTL
55/// Maximum memory for embedding cache (50MB)
56const EMBEDDING_CACHE_MAX_BYTES: usize = 50 * 1024 * 1024;

58/// Cached embedding with timestamp and size tracking
59#[derive(Clone)]
60struct CachedEmbedding {
61    embedding: Vec<f32>,
62    created_at: Instant,
63    /// Size in bytes (embedding.len() * 4 for f32)
64    size_bytes: usize,
65}
66
67impl CachedEmbedding {
68    fn new(embedding: Vec<f32>) -> Self {
69        let size_bytes = embedding.len() * std::mem::size_of::<f32>();
70        Self {
71            embedding,
72            created_at: Instant::now(),
73            size_bytes,
74        }
75    }
76
77    fn is_expired(&self) -> bool {
78        self.created_at.elapsed() > Duration::from_secs(EMBEDDING_CACHE_TTL_SECS)
79    }
80}

82/// Thread-safe embedding cache with LRU eviction and memory limits
83/// Uses VecDeque for O(1) front removal during eviction
84struct EmbeddingCache {
85    entries: HashMap<String, CachedEmbedding>,
86    access_order: VecDeque<String>, // For LRU tracking - O(1) pop_front
87    max_entries: usize,
88    max_bytes: usize,
89    current_bytes: usize,
90    hits: u64,
91    misses: u64,
92}
93
94impl EmbeddingCache {
95    fn new(max_entries: usize) -> Self {
96        Self {
97            entries: HashMap::with_capacity(max_entries.min(1000)), // Don't pre-allocate too much
98            access_order: VecDeque::with_capacity(max_entries.min(1000)),
99            max_entries,
100            max_bytes: EMBEDDING_CACHE_MAX_BYTES,
101            current_bytes: 0,
102            hits: 0,
103            misses: 0,
104        }
105    }
106
107    fn get(&mut self, key: &str) -> Option<Vec<f32>> {
108        if let Some(cached) = self.entries.get(key) {
109            if !cached.is_expired() {
110                self.hits += 1;
111                // Move to end for LRU - O(n) but only on cache hit
112                if let Some(pos) = self.access_order.iter().position(|k| k == key) {
113                    self.access_order.remove(pos);
114                    self.access_order.push_back(key.to_string());
115                }
116                return Some(cached.embedding.clone());
117            } else {
118                // Remove expired entry and update byte count
119                if let Some(removed) = self.entries.remove(key) {
120                    self.current_bytes = self.current_bytes.saturating_sub(removed.size_bytes);
121                }
122                self.access_order.retain(|k| k != key);
123            }
124        }
125        self.misses += 1;
126        None
127    }
128
129    fn insert(&mut self, key: String, embedding: Vec<f32>) {
130        let new_size = embedding.len() * std::mem::size_of::<f32>();
131
132        // Evict if at entry capacity or memory limit - O(1) with VecDeque::pop_front
133        while (self.entries.len() >= self.max_entries
134            || self.current_bytes + new_size > self.max_bytes)
135            && !self.access_order.is_empty()
136        {
137            if let Some(oldest) = self.access_order.pop_front() {
138                if let Some(removed) = self.entries.remove(&oldest) {
139                    self.current_bytes = self.current_bytes.saturating_sub(removed.size_bytes);
140                }
141            }
142        }
143
144        let cached = CachedEmbedding::new(embedding);
145        self.current_bytes += cached.size_bytes;
146        self.entries.insert(key.clone(), cached);
147        self.access_order.push_back(key);
148    }
149
150    fn stats(&self) -> (u64, u64, usize) {
151        (self.hits, self.misses, self.entries.len())
152    }
153
154    /// Get memory usage in bytes
155    #[allow(dead_code)] // Reserved for future memory monitoring
156    fn memory_bytes(&self) -> usize {
157        self.current_bytes
158    }
159
160    fn clear(&mut self) {
161        self.entries.clear();
162        self.access_order.clear();
163        self.current_bytes = 0;
164    }
165}

326    /// Embed an image file
327    ///
328    /// TODO(image-embeddings): This is a degenerate placeholder, NOT a real
329    /// image embedding. It writes raw downsampled RGB pixel values into the
330    /// vector; the 16x16 downsample only fills ~170 of the 512 dims (the
331    /// `i * 3 + 2 < len` guard stops early) and the dimension-info block at
332    /// `dim_idx = 256 * 3 = 768` is dead code (768 > IMAGE_EMBEDDING_DIM=512).
333    /// These vectors are not semantically comparable across images. Replace with
334    /// real CLIP ViT-B/32 (512-dim) embeddings via ONNX Runtime. Left as-is
335    /// (behavior unchanged) because a proper fix needs model/ONNX infra that is
336    /// unavailable in this environment. See report "Deferred" section.
337    async fn embed_image(&self, path: &Path) -> Result<Vec<f32>> {
338        debug!("Generating image embedding for: {:?}", path);
339
340        match image::open(path) {
341            Ok(img) => {
342                // Create a simple embedding from image statistics
343                let rgb = img.to_rgb8();
344                let (width, height) = rgb.dimensions();
345
346                // Calculate basic statistics per channel
347                let mut embedding = vec![0.0f32; IMAGE_EMBEDDING_DIM];
348
349                // Resize to 16x16 for feature extraction
350                let small =
351                    image::imageops::resize(&rgb, 16, 16, image::imageops::FilterType::Triangle);
352
353                // Use pixel values as features
354                for (i, pixel) in small.pixels().enumerate() {
355                    if i * 3 + 2 < embedding.len() {
356                        embedding[i * 3] = pixel[0] as f32 / 255.0;
357                        embedding[i * 3 + 1] = pixel[1] as f32 / 255.0;
358                        embedding[i * 3 + 2] = pixel[2] as f32 / 255.0;
359                    }
360                }
361
362                // Add dimension info
363                let dim_idx = 256 * 3;
364                if dim_idx + 2 < embedding.len() {
365                    embedding[dim_idx] = (width as f32).ln() / 10.0;
366                    embedding[dim_idx + 1] = (height as f32).ln() / 10.0;
367                    embedding[dim_idx + 2] = (width as f32 / height as f32).min(5.0) / 5.0;
368                }
369
370                // Normalize
371                Self::normalize(&mut embedding);
372                Ok(embedding)
373            }
374            Err(e) => {
375                warn!("Failed to load image for embedding: {}", e);
376                Ok(vec![0.0; IMAGE_EMBEDDING_DIM])
377            }
378        }
379    }

383        debug!("Generating code embedding for: {:?}", path);
384
385        let code = std::fs::read_to_string(path).unwrap_or_default();

387        // Prepend language context so the single text model sees the language.

391        // only as last resort) so code and text vectors are comparable.

393    }

427    /// Embed arbitrary text content for storage (document path).
428    ///
429    /// Prefers the configured semantic backend (Ollama / OpenAI) and only falls
430    /// back to the deterministic hash embedding as a genuine last resort. This
431    /// is the SAME backend the batch path uses, so single-item and batch
432    /// embeddings always land in one comparable vector space. The result is
433    /// L2-normalized exactly once.
434    async fn embed_text_content(&self, text: &str) -> Result<Vec<f32>> {
435        // Prefer Ollama (document task prefix applied inside embed_documents).
436        if let Some(ollama) = &self.ollama {
437            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
438                match ollama
439                    .embed_documents(std::slice::from_ref(&text.to_string()))
440                    .await
441                {
442                    Ok(mut embs) if !embs.is_empty() => {
443                        return Ok(Self::normalized(embs.remove(0)));
444                    }
445                    Ok(_) => warn!("Ollama returned no document embedding, falling back to hash"),
446                    Err(e) => warn!(
447                        "Ollama document embedding failed, falling back to hash: {}",
448                        e
449                    ),
450                }
451            }
452        }
453
454        // OpenAI API if configured.
455        if let Some(api_key) = &self.config.api_key {
456            if matches!(self.config.api_provider, EmbeddingProvider::OpenAI) {
457                return Ok(Self::normalized(
458                    self.embed_with_openai(text, api_key).await?,
459                ));
460            }
461        }
462
463        // Last resort: deterministic hash embedding (already normalized).
464        Ok(self.hash_embed(text, self.text_output_dim()))
465    }

467    /// Split a long document into sentence/paragraph-aware chunks with overlap.
468    ///
469    /// Tokens are approximated by whitespace word count (no tokenizer dependency).
470    /// Chunks target `chunk_tokens` words, carry `overlap_tokens` words over from
471    /// the previous chunk, and are never cut mid-word — every boundary falls on a
472    /// word boundary, and chunk ends prefer the nearest sentence/paragraph
473    /// boundary when one lands reasonably close to the size cap. Each returned
474    /// [`TextChunk`] carries BYTE offsets such that
475    /// `&text[char_start..char_end] == chunk.text` (for ASCII these equal
476    /// character offsets).
477    ///
478    /// A short text (<= `chunk_tokens` words) yields exactly one chunk covering
479    /// the whole text; whitespace-only / empty text yields no chunks.
480    pub fn chunk_document(
481        text: &str,
482        chunk_tokens: usize,
483        overlap_tokens: usize,
484    ) -> Vec<TextChunk> {
485        let chunk_tokens = chunk_tokens.max(1);
486        // Overlap must be strictly smaller than the chunk size to guarantee the
487        // window advances every iteration (otherwise the loop could not progress).
488        let overlap_tokens = overlap_tokens.min(chunk_tokens.saturating_sub(1));
489
490        // Byte span (start, end) of every whitespace-delimited word. Using byte
491        // offsets makes `&text[start..end]` slice exactly, so chunks round-trip.
492        let mut words: Vec<(usize, usize)> = Vec::new();
493        let mut word_start: Option<usize> = None;
494        for (i, ch) in text.char_indices() {
495            if ch.is_whitespace() {
496                if let Some(s) = word_start.take() {
497                    words.push((s, i));
498                }
499            } else if word_start.is_none() {
500                word_start = Some(i);
501            }
502        }
503        if let Some(s) = word_start {
504            words.push((s, text.len()));
505        }
506
507        let n = words.len();
508        if n == 0 {
509            return Vec::new();
510        }
511
512        // `sentence_end[i]` marks word `i` as the last word of a sentence or
513        // paragraph: it ends with terminal punctuation (ignoring trailing
514        // quotes/brackets), the gap to the next word contains a newline, or it is
515        // the final word of the text.
516        let bytes = text.as_bytes();
517        let mut sentence_end = vec![false; n];
518        for i in 0..n {
519            let (start, end) = words[i];
520            let word_str = &text[start..end];
521            let trimmed = word_str.trim_end_matches(['"', '\'', ')', ']', '}', '»', '”', '’']);
522            let ends_punct =
523                trimmed.ends_with('.') || trimmed.ends_with('!') || trimmed.ends_with('?');
524            let gap_has_newline = if i + 1 < n {
525                bytes[end..words[i + 1].0].contains(&b'\n')
526            } else {
527                true
528            };
529            sentence_end[i] = ends_punct || gap_has_newline;
530        }
531
532        // Don't let a sentence-boundary trim produce a chunk smaller than half the
533        // target; below that we keep the hard word cap instead.
534        let min_chunk = (chunk_tokens / 2).max(1);
535
536        let mut chunks = Vec::new();
537        let mut start = 0usize;
538        loop {
539            let hard_end = (start + chunk_tokens).min(n);
540            let mut end = hard_end;
541
542            if hard_end < n {
543                // Prefer the last sentence boundary in [start + min_chunk, hard_end).
544                let lower = start + min_chunk;
545                let mut w = hard_end;
546                while w > lower {
547                    w -= 1;
548                    if sentence_end[w] {
549                        end = w + 1; // `end` is exclusive
550                        break;
551                    }
552                }
553            }
554
555            let char_start = words[start].0;
556            let char_end = words[end - 1].1;
557            chunks.push(TextChunk {
558                text: text[char_start..char_end].to_string(),
559                char_start,
560                char_end,
561            });
562
563            if end >= n {
564                break;
565            }
566
567            // Start the next chunk `overlap_tokens` words before this one ended,
568            // but always advance by at least one word.
569            let mut next_start = end.saturating_sub(overlap_tokens);
570            if next_start <= start {
571                next_start = start + 1;
572            }
573            start = next_start;
574        }
575
576        chunks
577    }

579    /// Embed a search query (with caching for repeated queries)
580    pub async fn embed_query(&self, query: &str) -> Result<Vec<f32>> {
581        // Check cache first. Key includes model id + dim so that switching
582        // models/providers never returns a stale vector from a prior model.
583        let cache_key = format!(
584            "query:{}:{}:{}",
585            self.model_id(),
586            self.text_output_dim(),
587            query
588        );
589        {
590            let mut cache = self.cache.write();
591            if let Some(cached) = cache.get(&cache_key) {
592                debug!("Embedding cache hit for query: {}", query);
593                return Ok(cached);
594            }
595        }
596
597        // Generate embedding
598        let embedding = self.embed_query_uncached(query).await?;
599
600        // Cache the result
601        {
602            let mut cache = self.cache.write();
603            cache.insert(cache_key, embedding.clone());
604        }
605
606        Ok(embedding)
607    }

609    /// Embed a search query without caching (internal use)
610    async fn embed_query_uncached(&self, query: &str) -> Result<Vec<f32>> {
611        // Use Ollama if available (query task prefix applied inside embed_query).
612        if let Some(ollama) = &self.ollama {
613            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
614                match ollama.embed_query(query).await {
615                    Ok(embedding) => return Ok(Self::normalized(embedding)),
616                    Err(e) => {
617                        warn!("Ollama embedding failed, falling back to hash: {}", e);
618                    }
619                }
620            }
621        }
622
623        // Use OpenAI API if available
624        if let Some(api_key) = &self.config.api_key {
625            if matches!(self.config.api_provider, EmbeddingProvider::OpenAI) {
626                return Ok(Self::normalized(
627                    self.embed_with_openai(query, api_key).await?,
628                ));
629            }
630        }
631
632        // Fallback to hash-based embedding (already normalized).
633        Ok(self.hash_embed(query, self.text_output_dim()))
634    }

647    /// Embed multiple texts (document path) using the configured backend.
648    /// Every returned vector is L2-normalized exactly once.
649    pub async fn embed_batch(&self, texts: &[String]) -> Result<Vec<Vec<f32>>> {
650        if let Some(ollama) = &self.ollama {
651            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
652                match ollama.embed_documents(texts).await {
653                    Ok(embs) => return Ok(embs.into_iter().map(Self::normalized).collect()),
654                    Err(e) => warn!("Ollama batch embedding failed, falling back to hash: {}", e),
655                }
656            }
657        }
658        // Fallback to individual hash embeddings (already normalized).
659        Ok(texts
660            .iter()
661            .map(|t| self.hash_embed(t, self.text_output_dim()))
662            .collect())
663    }

665    /// Batch embed multiple memories efficiently
666    /// Groups memories by type and uses batch API for text-based content
667    pub async fn embed_memories_batch(
668        &self,
669        memories: &[Memory],
670    ) -> Result<Vec<(MemoryId, Vec<f32>, &'static str)>> {
671        let mut results: Vec<(MemoryId, Vec<f32>, &'static str)> =
672            Vec::with_capacity(memories.len());
673
674        // Separate memories by type for optimized processing
675        let mut text_memories: Vec<(usize, &Memory)> = Vec::new();
676        let mut image_memories: Vec<(usize, &Memory)> = Vec::new();
677        let mut code_memories: Vec<(usize, &Memory)> = Vec::new();
678
679        for (idx, memory) in memories.iter().enumerate() {
680            match &memory.kind {
681                MemoryKind::Image { .. } => image_memories.push((idx, memory)),
682                MemoryKind::Code { .. } => code_memories.push((idx, memory)),
683                _ => text_memories.push((idx, memory)),
684            }
685        }
686
687        // Process images individually (local computation, fast)
688        for (_idx, memory) in &image_memories {
689            match self.embed_image(&memory.path).await {
690                Ok(embedding) => {
691                    results.push((memory.id, embedding, "image_embedding"));
692                }
693                Err(e) => {
694                    warn!("Failed to embed image {}: {}", memory.id, e);
695                }
696            }
697        }
698
699        // Batch process code files if we have Ollama, otherwise use hash
700        if !code_memories.is_empty() {
701            let code_texts: Vec<String> = code_memories
702                .iter()
703                .map(|(_, m)| {
704                    std::fs::read_to_string(&m.path)
705                        .unwrap_or_else(|_| m.path.display().to_string())
706                        .chars()
707                        .take(4000) // Limit for efficiency
708                        .collect()
709                })
710                .collect();
711
712            if let Some(ollama) = &self.ollama {
713                // Batch embed code with Ollama (document task prefix applied).
714                match ollama.embed_documents(&code_texts).await {
715                    Ok(embeddings) => {
716                        for (embedding, (_, memory)) in
717                            embeddings.into_iter().zip(code_memories.iter())
718                        {
719                            results.push((
720                                memory.id,
721                                Self::normalized(embedding),
722                                "code_embedding",
723                            ));
724                        }
725                    }
726                    Err(e) => {
727                        warn!("Ollama batch code embedding failed, using fallback: {}", e);
728                        for (text, (_, memory)) in code_texts.iter().zip(code_memories.iter()) {
729                            results.push((
730                                memory.id,
731                                self.hash_embed(text, CODE_EMBEDDING_DIM),
732                                "code_embedding",
733                            ));
734                        }
735                    }
736                }
737            } else {
738                // Fallback to hash embeddings
739                for (text, (_, memory)) in code_texts.iter().zip(code_memories.iter()) {
740                    results.push((
741                        memory.id,
742                        self.hash_embed(text, CODE_EMBEDDING_DIM),
743                        "code_embedding",
744                    ));
745                }
746            }
747        }
748
749        // Batch process text/document memories
750        if !text_memories.is_empty() {
751            let texts: Vec<String> = text_memories
752                .iter()
753                .map(|(_, m)| {
754                    // Get text content for embedding
755                    let tags: Vec<String> = m.tags.iter().map(|t| t.name.clone()).collect();
756                    let content = std::fs::read_to_string(&m.path)
757                        .ok()
758                        .map(|s| s.chars().take(4000).collect::<String>())
759                        .unwrap_or_default();
760
761                    format!("{} {} {}", m.path.display(), tags.join(" "), content)
762                })
763                .collect();
764
765            if let Some(ollama) = &self.ollama {
766                // Batch embed with Ollama (document task prefix applied).
767                match ollama.embed_documents(&texts).await {
768                    Ok(embeddings) => {
769                        for (embedding, (_, memory)) in
770                            embeddings.into_iter().zip(text_memories.iter())
771                        {
772                            results.push((
773                                memory.id,
774                                Self::normalized(embedding),
775                                "text_embedding",
776                            ));
777                        }
778                    }
779                    Err(e) => {
780                        warn!("Ollama batch text embedding failed, using fallback: {}", e);
781                        for (text, (_, memory)) in texts.iter().zip(text_memories.iter()) {
782                            results.push((
783                                memory.id,
784                                self.hash_embed(text, TEXT_EMBEDDING_DIM),
785                                "text_embedding",
786                            ));
787                        }
788                    }
789                }
790            } else {
791                // Fallback to hash embeddings
792                for (text, (_, memory)) in texts.iter().zip(text_memories.iter()) {
793                    results.push((
794                        memory.id,
795                        self.hash_embed(text, TEXT_EMBEDDING_DIM),
796                        "text_embedding",
797                    ));
798                }
799            }
800        }
801
802        debug!(
803            "Batch embedded {} memories ({} images, {} code, {} text)",
804            results.len(),
805            image_memories.len(),
806            code_memories.len(),
807            text_memories.len()
808        );
809
810        Ok(results)
811    }

813    /// Embed text using OpenAI API
814    async fn embed_with_openai(&self, text: &str, api_key: &str) -> Result<Vec<f32>> {
815        #[derive(Serialize)]
816        struct EmbedRequest {
817            model: String,
818            input: String,
819        }
820
821        #[derive(Deserialize)]
822        struct EmbedResponse {
823            data: Vec<EmbedData>,
824        }
825
826        #[derive(Deserialize)]
827        struct EmbedData {
828            embedding: Vec<f32>,
829        }
830
831        // Truncate text to avoid token limits
832        let truncated: String = text.chars().take(8000).collect();
833
834        let request = EmbedRequest {
835            // text-embedding-ada-002 is deprecated; text-embedding-3-small is
836            // the current small model (also 1536-dim, cheaper, higher quality).
837            model: "text-embedding-3-small".to_string(),
838            input: truncated,
839        };
840
841        let response = self
842            .client
843            .post("https://api.openai.com/v1/embeddings")
844            .header("Authorization", format!("Bearer {}", api_key))
845            .header("Content-Type", "application/json")
846            .json(&request)
847            .send()
848            .await
849            .map_err(|e| HippoError::Other(format!("OpenAI API error: {}", e)))?;
850
851        if !response.status().is_success() {
852            let status = response.status();
853            warn!(
854                "OpenAI embedding failed with status {}, falling back to local",
855                status
856            );
857            // Keep the fallback in the OpenAI space dimension so it matches the
858            // collection created for the OpenAI provider.
859            return Ok(self.hash_embed(text, OPENAI_EMBEDDING_DIM));
860        }
861
862        let embed_response: EmbedResponse = response
863            .json()
864            .await
865            .map_err(|e| HippoError::Other(format!("Failed to parse embedding response: {}", e)))?;
866
867        embed_response
868            .data
869            .first()
870            .map(|d| d.embedding.clone())
871            .ok_or_else(|| HippoError::Other("No embedding in response".to_string()))
872    }

874    /// Simple hash-based embedding (fallback for offline use)
875    fn hash_embed(&self, text: &str, dim: usize) -> Vec<f32> {
876        let mut embedding = vec![0.0f32; dim];
877
878        // Tokenize and hash
879        let tokens: Vec<&str> = text.split_whitespace().collect();
880        let total_tokens = tokens.len().max(1) as f32;
881
882        for (pos, token) in tokens.iter().enumerate() {
883            let token_lower = token.to_lowercase();
884            let hash = Self::simple_hash(&token_lower);
885
886            // Position-weighted hashing
887            let position_weight = 1.0 - (pos as f32 / total_tokens) * 0.5;
888
889            // Multiple hash positions for better distribution
890            for i in 0..3 {
891                let idx = ((hash as usize).wrapping_add(i * 7919)) % dim;
892                embedding[idx] += position_weight * (1.0 / (i + 1) as f32);
893            }
894
895            // Bigram features
896            if pos > 0 {
897                let bigram = format!("{}_{}", tokens[pos - 1].to_lowercase(), token_lower);
898                let bigram_hash = Self::simple_hash(&bigram);
899                let idx = (bigram_hash as usize) % dim;
900                embedding[idx] += position_weight * 0.5;
901            }
902        }
903
904        // Character n-gram features for better matching
905        let chars: Vec<char> = text.to_lowercase().chars().collect();
906        for window in chars.windows(3) {
907            let ngram: String = window.iter().collect();
908            let hash = Self::simple_hash(&ngram);
909            let idx = (hash as usize) % dim;
910            embedding[idx] += 0.1;
911        }
912
913        Self::normalize(&mut embedding);
914        embedding
915    }
916
917    /// Simple string hash
918    fn simple_hash(s: &str) -> u64 {
919        let mut hash: u64 = 5381;
920        for c in s.chars() {
921            hash = hash.wrapping_mul(33).wrapping_add(c as u64);
922        }
923        hash
924    }

926    /// Normalize embedding to unit length
927    fn normalize(embedding: &mut [f32]) {
928        let norm: f32 = embedding.iter().map(|x| x * x).sum::<f32>().sqrt();
929        if norm > 0.0 {
930            for x in embedding.iter_mut() {
931                *x /= norm;
932            }
933        }
934    }

936    /// Canonical cosine similarity between two embeddings.
937    ///
938    /// All embeddings produced by this module are L2-normalized exactly once at
939    /// generation time, so for such vectors the dot product IS the cosine
940    /// similarity. Returns 0.0 on dimension mismatch.
941    ///
942    /// NOTE: `search::semantic_score`, `storage::cosine_similarity` and
943    /// `duplicates::cosine_similarity` are near-duplicates of this. Collapsing
944    /// them into this one is a cross-module follow-up (those files are owned by
945    /// other agents / depend on true-magnitude cosine in existing tests).
946    pub fn similarity(a: &[f32], b: &[f32]) -> f32 {
947        if a.len() != b.len() {
948            return 0.0;
949        }
950        a.iter().zip(b.iter()).map(|(x, y)| x * y).sum()
951    }

953    /// Find most similar embeddings from a list
954    pub fn find_similar(
955        query: &[f32],
956        embeddings: &[(String, Vec<f32>)],
957        top_k: usize,
958    ) -> Vec<(String, f32)> {
959        let mut scored: Vec<(String, f32)> = embeddings
960            .iter()
961            .map(|(id, emb)| (id.clone(), Self::similarity(query, emb)))
962            .collect();
963
964        scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
965        scored.truncate(top_k);
966        scored
967    }
