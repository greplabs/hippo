21/// Embedding dimension for different model types.
22///
23/// IMPORTANT: `TEXT_EMBEDDING_DIM` MUST match the runtime output dim of the
24/// default local embedding model (Ollama `nomic-embed-text` → 768, see
25/// `crate::ollama::NOMIC_EMBED_DIM`). The hash fallback also emits this dim so
26/// that fallback vectors live in the same space as Ollama vectors. A regression
27/// here silently breaks vector storage (dim mismatch on Qdrant upsert).
28pub const TEXT_EMBEDDING_DIM: usize = 768; // nomic-embed-text (default local model)
29pub const IMAGE_EMBEDDING_DIM: usize = 512; // CLIP compatible
30pub const CODE_EMBEDDING_DIM: usize = 768; // CodeBERT compatible / nomic
31/// OpenAI text-embedding-3-small / ada-002 dimension (only used when the OpenAI
32/// backend is the active provider).
33pub const OPENAI_EMBEDDING_DIM: usize = 1536;

35/// Cache configuration
36const EMBEDDING_CACHE_SIZE: usize = 5000;
37const EMBEDDING_CACHE_TTL_SECS: u64 = 3600; // 1 hour TTL
38/// Maximum memory for embedding cache (50MB)
39const EMBEDDING_CACHE_MAX_BYTES: usize = 50 * 1024 * 1024;

41/// Cached embedding with timestamp and size tracking
42#[derive(Clone)]
43struct CachedEmbedding {
44    embedding: Vec<f32>,
45    created_at: Instant,
46    /// Size in bytes (embedding.len() * 4 for f32)
47    size_bytes: usize,
48}
49
50impl CachedEmbedding {
51    fn new(embedding: Vec<f32>) -> Self {
52        let size_bytes = embedding.len() * std::mem::size_of::<f32>();
53        Self {
54            embedding,
55            created_at: Instant::now(),
56            size_bytes,
57        }
58    }
59
60    fn is_expired(&self) -> bool {
61        self.created_at.elapsed() > Duration::from_secs(EMBEDDING_CACHE_TTL_SECS)
62    }
63}
64
65/// Thread-safe embedding cache with LRU eviction and memory limits
66/// Uses VecDeque for O(1) front removal during eviction
67struct EmbeddingCache {
68    entries: HashMap<String, CachedEmbedding>,
69    access_order: VecDeque<String>, // For LRU tracking - O(1) pop_front
70    max_entries: usize,
71    max_bytes: usize,
72    current_bytes: usize,
73    hits: u64,
74    misses: u64,
75}
76
77impl EmbeddingCache {
78    fn new(max_entries: usize) -> Self {
79        Self {
80            entries: HashMap::with_capacity(max_entries.min(1000)), // Don't pre-allocate too much
81            access_order: VecDeque::with_capacity(max_entries.min(1000)),
82            max_entries,
83            max_bytes: EMBEDDING_CACHE_MAX_BYTES,
84            current_bytes: 0,
85            hits: 0,
86            misses: 0,
87        }
88    }
89
90    fn get(&mut self, key: &str) -> Option<Vec<f32>> {
91        if let Some(cached) = self.entries.get(key) {
92            if !cached.is_expired() {
93                self.hits += 1;
94                // Move to end for LRU - O(n) but only on cache hit
95                if let Some(pos) = self.access_order.iter().position(|k| k == key) {
96                    self.access_order.remove(pos);
97                    self.access_order.push_back(key.to_string());
98                }
99                return Some(cached.embedding.clone());
100            } else {
101                // Remove expired entry and update byte count
102                if let Some(removed) = self.entries.remove(key) {
103                    self.current_bytes = self.current_bytes.saturating_sub(removed.size_bytes);
104                }
105                self.access_order.retain(|k| k != key);
106            }
107        }
108        self.misses += 1;
109        None
110    }
111
112    fn insert(&mut self, key: String, embedding: Vec<f32>) {
113        let new_size = embedding.len() * std::mem::size_of::<f32>();
114
115        // Evict if at entry capacity or memory limit - O(1) with VecDeque::pop_front
116        while (self.entries.len() >= self.max_entries
117            || self.current_bytes + new_size > self.max_bytes)
118            && !self.access_order.is_empty()
119        {
120            if let Some(oldest) = self.access_order.pop_front() {
121                if let Some(removed) = self.entries.remove(&oldest) {
122                    self.current_bytes = self.current_bytes.saturating_sub(removed.size_bytes);
123                }
124            }
125        }
126
127        let cached = CachedEmbedding::new(embedding);
128        self.current_bytes += cached.size_bytes;
129        self.entries.insert(key.clone(), cached);
130        self.access_order.push_back(key);
131    }
132
133    fn stats(&self) -> (u64, u64, usize) {
134        (self.hits, self.misses, self.entries.len())
135    }
136
137    /// Get memory usage in bytes
138    #[allow(dead_code)] // Reserved for future memory monitoring
139    fn memory_bytes(&self) -> usize {
140        self.current_bytes
141    }
142
143    fn clear(&mut self) {
144        self.entries.clear();
145        self.access_order.clear();
146        self.current_bytes = 0;
147    }
148}

260    /// Runtime output dimension of the active TEXT embedding backend.
261    ///
262    /// Qdrant collections should be created/validated against THIS value rather
263    /// than a hardcoded constant, so that the stored collection dim always
264    /// matches what the embedder actually emits.
265    pub fn text_output_dim(&self) -> usize {
266        match self.config.api_provider {
267            EmbeddingProvider::OpenAI => OPENAI_EMBEDDING_DIM,
268            // Ollama `nomic-embed-text` and the hash fallback both emit TEXT_EMBEDDING_DIM.
269            EmbeddingProvider::Ollama | EmbeddingProvider::Local => TEXT_EMBEDDING_DIM,
270        }

273    /// Stable identifier of the active embedding model (provider + model + dim).
274    ///
275    /// Used in cache keys so that switching models/providers never returns a
276    /// stale vector from a previous model.
277    pub fn model_id(&self) -> String {
278        match self.config.api_provider {
279            EmbeddingProvider::Ollama => self
280                .ollama
281                .as_ref()
282                .map(|o| o.embedding_model().to_string())
283                .unwrap_or_else(|| "ollama".to_string()),
284            EmbeddingProvider::OpenAI => "openai:text-embedding-3-small".to_string(),
285            EmbeddingProvider::Local => "hash".to_string(),
286        }

289    /// L2-normalize a freshly generated embedding exactly once.
290    ///
291    /// Call this at the single point where a vector is produced by a backend
292    /// (Ollama / OpenAI). After normalization, dot product == cosine similarity,
293    /// which is what `Embedder::similarity` and Qdrant's cosine distance assume.
294    fn normalized(mut embedding: Vec<f32>) -> Vec<f32> {
295        Self::normalize(&mut embedding);
296        embedding
297    }

299    /// Generate embedding for a memory
300    pub async fn embed_memory(&self, memory: &Memory) -> Result<Vec<f32>> {
301        match &memory.kind {
302            MemoryKind::Image { .. } => self.embed_image(&memory.path).await,
303            MemoryKind::Code { language, .. } => self.embed_code(&memory.path, language).await,
304            MemoryKind::Document { .. } => self.embed_document(&memory.path).await,
305            _ => self.embed_generic(memory).await,
306        }
307    }

309    /// Embed an image file
310    ///
311    /// TODO(image-embeddings): This is a degenerate placeholder, NOT a real
312    /// image embedding. It writes raw downsampled RGB pixel values into the
313    /// vector; the 16x16 downsample only fills ~170 of the 512 dims (the
314    /// `i * 3 + 2 < len` guard stops early) and the dimension-info block at
315    /// `dim_idx = 256 * 3 = 768` is dead code (768 > IMAGE_EMBEDDING_DIM=512).
316    /// These vectors are not semantically comparable across images. Replace with
317    /// real CLIP ViT-B/32 (512-dim) embeddings via ONNX Runtime. Left as-is
318    /// (behavior unchanged) because a proper fix needs model/ONNX infra that is
319    /// unavailable in this environment. See report "Deferred" section.
320    async fn embed_image(&self, path: &Path) -> Result<Vec<f32>> {
321        debug!("Generating image embedding for: {:?}", path);
322
323        match image::open(path) {
324            Ok(img) => {
325                // Create a simple embedding from image statistics
326                let rgb = img.to_rgb8();
327                let (width, height) = rgb.dimensions();
328
329                // Calculate basic statistics per channel
330                let mut embedding = vec![0.0f32; IMAGE_EMBEDDING_DIM];
331
332                // Resize to 16x16 for feature extraction
333                let small =
334                    image::imageops::resize(&rgb, 16, 16, image::imageops::FilterType::Triangle);
335
336                // Use pixel values as features
337                for (i, pixel) in small.pixels().enumerate() {
338                    if i * 3 + 2 < embedding.len() {
339                        embedding[i * 3] = pixel[0] as f32 / 255.0;
340                        embedding[i * 3 + 1] = pixel[1] as f32 / 255.0;
341                        embedding[i * 3 + 2] = pixel[2] as f32 / 255.0;
342                    }
343                }
344
345                // Add dimension info
346                let dim_idx = 256 * 3;
347                if dim_idx + 2 < embedding.len() {
348                    embedding[dim_idx] = (width as f32).ln() / 10.0;
349                    embedding[dim_idx + 1] = (height as f32).ln() / 10.0;
350                    embedding[dim_idx + 2] = (width as f32 / height as f32).min(5.0) / 5.0;
351                }
352
353                // Normalize
354                Self::normalize(&mut embedding);
355                Ok(embedding)
356            }
357            Err(e) => {
358                warn!("Failed to load image for embedding: {}", e);
359                Ok(vec![0.0; IMAGE_EMBEDDING_DIM])
360            }
361        }
362    }

364    /// Embed a code file
365    async fn embed_code(&self, path: &Path, language: &str) -> Result<Vec<f32>> {
366        debug!("Generating code embedding for: {:?}", path);
367
368        let code = std::fs::read_to_string(path).unwrap_or_default();
369        let truncated: String = code.chars().take(4000).collect();
370        // Prepend language context so the single text model sees the language.
371        let content = format!("{} source code\n{}", language, truncated);
372
373        // Share the SAME backend/space as the batch path (Ollama/OpenAI, hash
374        // only as last resort) so code and text vectors are comparable.
375        self.embed_text_content(&content).await
376    }

378    /// Embed a document file
379    async fn embed_document(&self, path: &Path) -> Result<Vec<f32>> {
380        debug!("Generating document embedding for: {:?}", path);
381
382        // Try to read text content
383        let text = match std::fs::read_to_string(path) {
384            Ok(t) => t.chars().take(4000).collect::<String>(),
385            Err(_) => {
386                // For binary documents, use filename and metadata
387                path.file_name()
388                    .map(|n| n.to_string_lossy().to_string())
389                    .unwrap_or_default()
390            }
391        };
392
393        self.embed_text_content(&text).await
394    }

410    /// Embed arbitrary text content for storage (document path).
411    ///
412    /// Prefers the configured semantic backend (Ollama / OpenAI) and only falls
413    /// back to the deterministic hash embedding as a genuine last resort. This
414    /// is the SAME backend the batch path uses, so single-item and batch
415    /// embeddings always land in one comparable vector space. The result is
416    /// L2-normalized exactly once.
417    async fn embed_text_content(&self, text: &str) -> Result<Vec<f32>> {
418        // Prefer Ollama (document task prefix applied inside embed_documents).
419        if let Some(ollama) = &self.ollama {
420            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
421                match ollama
422                    .embed_documents(std::slice::from_ref(&text.to_string()))
423                    .await
424                {
425                    Ok(mut embs) if !embs.is_empty() => {
426                        return Ok(Self::normalized(embs.remove(0)));
427                    }
428                    Ok(_) => warn!("Ollama returned no document embedding, falling back to hash"),
429                    Err(e) => warn!(
430                        "Ollama document embedding failed, falling back to hash: {}",
431                        e
432                    ),
433                }
434            }
435        }
436
437        // OpenAI API if configured.
438        if let Some(api_key) = &self.config.api_key {
439            if matches!(self.config.api_provider, EmbeddingProvider::OpenAI) {
440                return Ok(Self::normalized(
441                    self.embed_with_openai(text, api_key).await?,
442                ));
443            }
444        }
445
446        // Last resort: deterministic hash embedding (already normalized).
447        Ok(self.hash_embed(text, self.text_output_dim()))
448    }

465    /// Embed a search query (with caching for repeated queries)
466    pub async fn embed_query(&self, query: &str) -> Result<Vec<f32>> {
467        // Check cache first. Key includes model id + dim so that switching
468        // models/providers never returns a stale vector from a prior model.
469        let cache_key = format!(
470            "query:{}:{}:{}",
471            self.model_id(),
472            self.text_output_dim(),
473            query
474        );
475        {
476            let mut cache = self.cache.write();
477            if let Some(cached) = cache.get(&cache_key) {
478                debug!("Embedding cache hit for query: {}", query);
479                return Ok(cached);
480            }
481        }
482
483        // Generate embedding
484        let embedding = self.embed_query_uncached(query).await?;
485
486        // Cache the result
487        {
488            let mut cache = self.cache.write();
489            cache.insert(cache_key, embedding.clone());
490        }
491
492        Ok(embedding)
493    }

495    /// Embed a search query without caching (internal use)
496    async fn embed_query_uncached(&self, query: &str) -> Result<Vec<f32>> {
497        // Use Ollama if available (query task prefix applied inside embed_query).
498        if let Some(ollama) = &self.ollama {
499            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
500                match ollama.embed_query(query).await {
501                    Ok(embedding) => return Ok(Self::normalized(embedding)),
502                    Err(e) => {
503                        warn!("Ollama embedding failed, falling back to hash: {}", e);
504                    }
505                }
506            }
507        }
508
509        // Use OpenAI API if available
510        if let Some(api_key) = &self.config.api_key {
511            if matches!(self.config.api_provider, EmbeddingProvider::OpenAI) {
512                return Ok(Self::normalized(
513                    self.embed_with_openai(query, api_key).await?,
514                ));
515            }
516        }
517
518        // Fallback to hash-based embedding (already normalized).
519        Ok(self.hash_embed(query, self.text_output_dim()))
520    }

533    /// Embed multiple texts (document path) using the configured backend.
534    /// Every returned vector is L2-normalized exactly once.
535    pub async fn embed_batch(&self, texts: &[String]) -> Result<Vec<Vec<f32>>> {
536        if let Some(ollama) = &self.ollama {
537            if matches!(self.config.api_provider, EmbeddingProvider::Ollama) {
538                match ollama.embed_documents(texts).await {
539                    Ok(embs) => return Ok(embs.into_iter().map(Self::normalized).collect()),
540                    Err(e) => warn!("Ollama batch embedding failed, falling back to hash: {}", e),
541                }
542            }
543        }
544        // Fallback to individual hash embeddings (already normalized).
545        Ok(texts
546            .iter()
547            .map(|t| self.hash_embed(t, self.text_output_dim()))
548            .collect())
549    }

551    /// Batch embed multiple memories efficiently
552    /// Groups memories by type and uses batch API for text-based content
553    pub async fn embed_memories_batch(
554        &self,
555        memories: &[Memory],
556    ) -> Result<Vec<(MemoryId, Vec<f32>, &'static str)>> {
557        let mut results: Vec<(MemoryId, Vec<f32>, &'static str)> =
558            Vec::with_capacity(memories.len());
559
560        // Separate memories by type for optimized processing
561        let mut text_memories: Vec<(usize, &Memory)> = Vec::new();
562        let mut image_memories: Vec<(usize, &Memory)> = Vec::new();
563        let mut code_memories: Vec<(usize, &Memory)> = Vec::new();
564
565        for (idx, memory) in memories.iter().enumerate() {
566            match &memory.kind {
567                MemoryKind::Image { .. } => image_memories.push((idx, memory)),
568                MemoryKind::Code { .. } => code_memories.push((idx, memory)),
569                _ => text_memories.push((idx, memory)),
570            }
571        }
572
573        // Process images individually (local computation, fast)
574        for (_idx, memory) in &image_memories {
575            match self.embed_image(&memory.path).await {
576                Ok(embedding) => {
577                    results.push((memory.id, embedding, "image_embedding"));
578                }
579                Err(e) => {
580                    warn!("Failed to embed image {}: {}", memory.id, e);
581                }
582            }
583        }
584
585        // Batch process code files if we have Ollama, otherwise use hash
586        if !code_memories.is_empty() {
587            let code_texts: Vec<String> = code_memories
588                .iter()
589                .map(|(_, m)| {
590                    std::fs::read_to_string(&m.path)
591                        .unwrap_or_else(|_| m.path.display().to_string())
592                        .chars()
593                        .take(4000) // Limit for efficiency
594                        .collect()
595                })
596                .collect();
597
598            if let Some(ollama) = &self.ollama {
599                // Batch embed code with Ollama (document task prefix applied).
600                match ollama.embed_documents(&code_texts).await {
601                    Ok(embeddings) => {
602                        for (embedding, (_, memory)) in
603                            embeddings.into_iter().zip(code_memories.iter())
604                        {
605                            results.push((
606                                memory.id,
607                                Self::normalized(embedding),
608                                "code_embedding",
609                            ));
610                        }
611                    }
612                    Err(e) => {
613                        warn!("Ollama batch code embedding failed, using fallback: {}", e);
614                        for (text, (_, memory)) in code_texts.iter().zip(code_memories.iter()) {
615                            results.push((
616                                memory.id,
617                                self.hash_embed(text, CODE_EMBEDDING_DIM),
618                                "code_embedding",
619                            ));
620                        }
621                    }
622                }
623            } else {
624                // Fallback to hash embeddings
625                for (text, (_, memory)) in code_texts.iter().zip(code_memories.iter()) {
626                    results.push((
627                        memory.id,
628                        self.hash_embed(text, CODE_EMBEDDING_DIM),
629                        "code_embedding",
630                    ));
631                }
632            }
633        }
634
635        // Batch process text/document memories
636        if !text_memories.is_empty() {
637            let texts: Vec<String> = text_memories
638                .iter()
639                .map(|(_, m)| {
640                    // Get text content for embedding
641                    let tags: Vec<String> = m.tags.iter().map(|t| t.name.clone()).collect();
642                    let content = std::fs::read_to_string(&m.path)
643                        .ok()
644                        .map(|s| s.chars().take(4000).collect::<String>())
645                        .unwrap_or_default();
646
647                    format!("{} {} {}", m.path.display(), tags.join(" "), content)
648                })
649                .collect();
650
651            if let Some(ollama) = &self.ollama {
652                // Batch embed with Ollama (document task prefix applied).
653                match ollama.embed_documents(&texts).await {
654                    Ok(embeddings) => {
655                        for (embedding, (_, memory)) in
656                            embeddings.into_iter().zip(text_memories.iter())
657                        {
658                            results.push((
659                                memory.id,
660                                Self::normalized(embedding),
661                                "text_embedding",
662                            ));
663                        }
664                    }
665                    Err(e) => {
666                        warn!("Ollama batch text embedding failed, using fallback: {}", e);
667                        for (text, (_, memory)) in texts.iter().zip(text_memories.iter()) {
668                            results.push((
669                                memory.id,
670                                self.hash_embed(text, TEXT_EMBEDDING_DIM),
671                                "text_embedding",
672                            ));
673                        }
674                    }
675                }
676            } else {
677                // Fallback to hash embeddings
678                for (text, (_, memory)) in texts.iter().zip(text_memories.iter()) {
679                    results.push((
680                        memory.id,
681                        self.hash_embed(text, TEXT_EMBEDDING_DIM),
682                        "text_embedding",
683                    ));
684                }
685            }
686        }
687
688        debug!(
689            "Batch embedded {} memories ({} images, {} code, {} text)",
690            results.len(),
691            image_memories.len(),
692            code_memories.len(),
693            text_memories.len()
694        );
695
696        Ok(results)
697    }

699    /// Embed text using OpenAI API
700    async fn embed_with_openai(&self, text: &str, api_key: &str) -> Result<Vec<f32>> {
701        #[derive(Serialize)]
702        struct EmbedRequest {
703            model: String,
704            input: String,
705        }
706
707        #[derive(Deserialize)]
708        struct EmbedResponse {
709            data: Vec<EmbedData>,
710        }
711
712        #[derive(Deserialize)]
713        struct EmbedData {
714            embedding: Vec<f32>,
715        }
716
717        // Truncate text to avoid token limits
718        let truncated: String = text.chars().take(8000).collect();
719
720        let request = EmbedRequest {
721            // text-embedding-ada-002 is deprecated; text-embedding-3-small is
722            // the current small model (also 1536-dim, cheaper, higher quality).
723            model: "text-embedding-3-small".to_string(),
724            input: truncated,
725        };
726
727        let response = self
728            .client
729            .post("https://api.openai.com/v1/embeddings")
730            .header("Authorization", format!("Bearer {}", api_key))
731            .header("Content-Type", "application/json")
732            .json(&request)
733            .send()
734            .await
735            .map_err(|e| HippoError::Other(format!("OpenAI API error: {}", e)))?;
736
737        if !response.status().is_success() {
738            let status = response.status();
739            warn!(
740                "OpenAI embedding failed with status {}, falling back to local",
741                status
742            );
743            // Keep the fallback in the OpenAI space dimension so it matches the
744            // collection created for the OpenAI provider.
745            return Ok(self.hash_embed(text, OPENAI_EMBEDDING_DIM));
746        }
747
748        let embed_response: EmbedResponse = response
749            .json()
750            .await
751            .map_err(|e| HippoError::Other(format!("Failed to parse embedding response: {}", e)))?;
752
753        embed_response
754            .data
755            .first()
756            .map(|d| d.embedding.clone())
757            .ok_or_else(|| HippoError::Other("No embedding in response".to_string()))
758    }

760    /// Simple hash-based embedding (fallback for offline use)
761    fn hash_embed(&self, text: &str, dim: usize) -> Vec<f32> {
762        let mut embedding = vec![0.0f32; dim];
763
764        // Tokenize and hash
765        let tokens: Vec<&str> = text.split_whitespace().collect();
766        let total_tokens = tokens.len().max(1) as f32;
767
768        for (pos, token) in tokens.iter().enumerate() {
769            let token_lower = token.to_lowercase();
770            let hash = Self::simple_hash(&token_lower);
771
772            // Position-weighted hashing
773            let position_weight = 1.0 - (pos as f32 / total_tokens) * 0.5;
774
775            // Multiple hash positions for better distribution
776            for i in 0..3 {
777                let idx = ((hash as usize).wrapping_add(i * 7919)) % dim;
778                embedding[idx] += position_weight * (1.0 / (i + 1) as f32);
779            }
780
781            // Bigram features
782            if pos > 0 {
783                let bigram = format!("{}_{}", tokens[pos - 1].to_lowercase(), token_lower);
784                let bigram_hash = Self::simple_hash(&bigram);
785                let idx = (bigram_hash as usize) % dim;
786                embedding[idx] += position_weight * 0.5;
787            }
788        }
789
790        // Character n-gram features for better matching
791        let chars: Vec<char> = text.to_lowercase().chars().collect();
792        for window in chars.windows(3) {
793            let ngram: String = window.iter().collect();
794            let hash = Self::simple_hash(&ngram);
795            let idx = (hash as usize) % dim;
796            embedding[idx] += 0.1;
797        }
798
799        Self::normalize(&mut embedding);
800        embedding
801    }

803    /// Simple string hash
804    fn simple_hash(s: &str) -> u64 {
805        let mut hash: u64 = 5381;
806        for c in s.chars() {
807            hash = hash.wrapping_mul(33).wrapping_add(c as u64);
808        }
809        hash
810    }

812    /// Normalize embedding to unit length
813    fn normalize(embedding: &mut [f32]) {
814        let norm: f32 = embedding.iter().map(|x| x * x).sum::<f32>().sqrt();
815        if norm > 0.0 {
816            for x in embedding.iter_mut() {
817                *x /= norm;
818            }
819        }
820    }

822    /// Canonical cosine similarity between two embeddings.
823    ///
824    /// All embeddings produced by this module are L2-normalized exactly once at
825    /// generation time, so for such vectors the dot product IS the cosine
826    /// similarity. Returns 0.0 on dimension mismatch.
827    ///
828    /// NOTE: `search::semantic_score`, `storage::cosine_similarity` and
829    /// `duplicates::cosine_similarity` are near-duplicates of this. Collapsing
830    /// them into this one is a cross-module follow-up (those files are owned by
831    /// other agents / depend on true-magnitude cosine in existing tests).
832    pub fn similarity(a: &[f32], b: &[f32]) -> f32 {
833        if a.len() != b.len() {
834            return 0.0;
835        }
836        a.iter().zip(b.iter()).map(|(x, y)| x * y).sum()
837    }
838
839    /// Find most similar embeddings from a list
840    pub fn find_similar(
841        query: &[f32],
842        embeddings: &[(String, Vec<f32>)],
843        top_k: usize,
844    ) -> Vec<(String, f32)> {
845        let mut scored: Vec<(String, f32)> = embeddings
846            .iter()
847            .map(|(id, emb)| (id.clone(), Self::similarity(query, emb)))
848            .collect();
849
850        scored.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
851        scored.truncate(top_k);
852        scored
853    }

864            embeddings: HashMap::new(),
865        }

868    /// Add an embedding to the index
869    pub fn add(&mut self, id: String, embedding: Vec<f32>) {
870        self.embeddings.insert(id, embedding);
871    }

874    pub fn remove(&mut self, id: &str) {
875        self.embeddings.remove(id);
876    }

878    /// Search for similar vectors
879    pub fn search(&self, query: &[f32], top_k: usize) -> Vec<(String, f32)> {
880        let embeddings_vec: Vec<(String, Vec<f32>)> = self
881            .embeddings
882            .iter()
883            .map(|(k, v)| (k.clone(), v.clone()))
884            .collect();
885
886        Embedder::find_similar(query, &embeddings_vec, top_k)
887    }

892    }
