1//! # Hippo Web API
2//!
3//! REST API server for Hippo - intelligent file memory system.
4//!
5//! ## Configuration
6//!
7//! - `HIPPO_HOST` - Server host (default: "127.0.0.1")
8//! - `HIPPO_PORT` - Server port (default: "3000")
9//! - `HIPPO_DATA_DIR` - Data directory for Hippo (optional)
10//!
11//! ## Endpoints
12//!
13//! - `GET /api/health` - Health check
14//! - `GET /api/stats` - Index statistics
15//! - `GET /api/search` - Search memories with filters
16//! - `GET /api/memories/:id` - Get single memory by ID
17//! - `GET /api/thumbnails/:id` - Get thumbnail image for memory
18//! - `GET /api/sources` - List configured sources
19//! - `POST /api/sources` - Add a new source
20//! - `DELETE /api/sources/:path` - Remove a source
21//! - `GET /api/tags` - List all tags with counts
22//! - `POST /api/memories/:id/tags` - Add tag to memory
23//! - `DELETE /api/memories/:id/tags/:tag` - Remove tag from memory
24

51#[tokio::main]
52async fn main() -> anyhow::Result<()> {
53    // Initialize tracing
54    tracing_subscriber::registry()
55        .with(
56            tracing_subscriber::EnvFilter::try_from_default_env()
57                .unwrap_or_else(|_| "hippo_web=info,tower_http=debug".into()),
58        )
59        .with(tracing_subscriber::fmt::layer())
60        .init();
61
62    // Load configuration from environment
63    let host = std::env::var("HIPPO_HOST").unwrap_or_else(|_| "127.0.0.1".to_string());
64    let port = std::env::var("HIPPO_PORT").unwrap_or_else(|_| "3000".to_string());
65
66    let mut config = HippoConfig::default();
67    if let Ok(data_dir) = std::env::var("HIPPO_DATA_DIR") {
68        config.data_dir = PathBuf::from(data_dir);
69    }
70
71    info!(
72        "Initializing Hippo with data directory: {:?}",
73        config.data_dir
74    );
75
76    // Initialize Hippo instance
77    let hippo = Arc::new(Hippo::with_config(config).await?);
78    info!("Hippo initialized successfully");
79
80    let state = AppState { hippo };
81
82    // Build router with CORS and tracing middleware
83    let app = build_router(state);
84
85    // Bind and serve
86    let addr: SocketAddr = format!("{}:{}", host, port).parse()?;
87    info!("Starting Hippo Web API server on http://{}", addr);
88
89    let listener = tokio::net::TcpListener::bind(&addr).await?;
90    axum::serve(listener, app).await?;
91
92    Ok(())
93}

95/// Build the application router. Extracted from `main` so tests can drive the
96/// real route table and handlers in-process via `tower::ServiceExt::oneshot`,
97/// without binding a TCP listener or starting a live server.
98fn build_router(state: AppState) -> Router {
99    Router::new()
100        // Health check
101        .route("/api/health", get(health_check))
102        // Stats
103        .route("/api/stats", get(get_stats))
104        // Search
105        .route("/api/search", get(search_memories))
106        // Memories
107        .route("/api/memories/:id", get(get_memory))
108        // Thumbnails
109        .route("/api/thumbnails/:id", get(get_thumbnail))
110        // Sources
111        .route("/api/sources", get(list_sources).post(add_source))
112        .route("/api/sources/*path", delete(remove_source))
113        // Tags
114        .route("/api/tags", get(list_tags))
115        .route("/api/memories/:id/tags", post(add_tag))
116        .route("/api/memories/:id/tags/:tag", delete(remove_tag))
117        // Apply middleware
118        .layer(
119            CorsLayer::new()
120                .allow_origin(Any)
121                .allow_methods(Any)
122                .allow_headers(Any),
123        )
124        .layer(TraceLayer::new_for_http())
125        .with_state(state)
126}

167/// GET /api/search
168///
169/// Search memories with optional filters and sorting.
170///
171/// ## Query Parameters
172/// - `q` (optional): Search text for semantic/keyword search
173/// - `tags` (optional): Comma-separated tags to filter by (prefix with `-` to exclude)
174/// - `type` (optional): Filter by memory kind (Image, Video, Audio, Code, Document)
175/// - `sort` (optional): Sort order (Relevance, DateNewest, DateOldest, NameAsc, NameDesc, SizeAsc, SizeDesc)
176/// - `limit` (optional): Maximum results to return (default: 50)
177/// - `offset` (optional): Offset for pagination (default: 0)
178/// - `mode` (optional): Search mode - "keyword" (default), "semantic", or "hybrid"
179///
180/// ## Example
181/// ```
182/// GET /api/search?q=sunset&tags=vacation,-private&type=Image&sort=DateNewest&limit=20
183/// GET /api/search?q=beautiful landscape&mode=hybrid&limit=20
184/// ```
185///
186/// ## Response
187/// Returns `SearchResults` containing:
188/// - `memories`: Array of matching memories with scores
189/// - `total_count`: Total matching results
190/// - `suggested_tags`: Tag suggestions based on results
191/// - `clusters`: Related clusters
192async fn search_memories(
193    State(state): State<AppState>,
194    Query(params): Query<SearchParams>,
195) -> Result<Json<SearchResults>, ApiError> {
196    let mode = params.mode.as_deref().unwrap_or("keyword");
197    let limit = params.limit.unwrap_or(50);
198
199    // Use different search methods based on mode
200    let results = match mode {
201        "hybrid" => {
202            // Hybrid search: 70% semantic + 30% keyword
203            if let Some(ref q) = params.q {
204                state.hippo.hybrid_search(q, limit).await?
205            } else {
206                let query = params.into_search_query()?;
207                state.hippo.search_advanced(query).await?
208            }
209        }
210        "semantic" => {
211            // Pure semantic/embedding search
212            if let Some(ref q) = params.q {
213                state.hippo.semantic_search(q, limit).await?
214            } else {
215                let query = params.into_search_query()?;
216                state.hippo.search_advanced(query).await?
217            }
218        }
219        _ => {
220            // Default: keyword-based search with filters
221            let query = params.into_search_query()?;
222            state.hippo.search_advanced(query).await?
223        }
224    };
225
226    Ok(Json(results))
227}

254/// GET /api/thumbnails/:id
255///
256/// Get thumbnail image for a memory (images and videos only).
257///
258/// ## Path Parameters
259/// - `id`: UUID of the memory
260///
261/// ## Response
262/// Returns JPEG image data with appropriate content-type header.
263/// Returns 404 if memory not found or thumbnail cannot be generated.
264async fn get_thumbnail(
265    State(state): State<AppState>,
266    Path(id): Path<String>,
267) -> Result<Response, ApiError> {
268    let memory_id = MemoryId::from_str(&id)
269        .map_err(|_| ApiError::BadRequest("Invalid memory ID".to_string()))?;
270
271    let memory = state
272        .hippo
273        .get_memory(memory_id)
274        .await?
275        .ok_or_else(|| ApiError::NotFound("Memory not found".to_string()))?;
276
277    // Try to get thumbnail based on memory kind
278    let thumbnail_path = match &memory.kind {
279        hippo_core::MemoryKind::Image { .. } => state.hippo.get_thumbnail(&memory.path).ok(),
280        hippo_core::MemoryKind::Video { .. } => state.hippo.get_video_thumbnail(&memory.path).ok(),
281        _ => None,
282    };
283
284    let thumbnail_path =
285        thumbnail_path.ok_or_else(|| ApiError::NotFound("Thumbnail not available".to_string()))?;
286
287    // Read thumbnail file
288    let data = tokio::fs::read(&thumbnail_path)
289        .await
290        .map_err(|e| ApiError::InternalError(format!("Failed to read thumbnail: {}", e)))?;
291
292    // Return image with proper content type
293    Ok(([(header::CONTENT_TYPE, "image/jpeg")], data).into_response())
294}

454#[derive(Debug, Deserialize)]
455struct SearchParams {
456    q: Option<String>,
457    tags: Option<String>,
458    #[serde(rename = "type")]
459    kind: Option<String>,
460    sort: Option<String>,
461    limit: Option<usize>,
462    offset: Option<usize>,
463    /// Search mode: "keyword" (default), "semantic", or "hybrid"
464    mode: Option<String>,
465}
466
467impl SearchParams {
468    fn into_search_query(self) -> Result<SearchQuery, ApiError> {
469        let mut query = SearchQuery {
470            text: self.q,
471            limit: self.limit.unwrap_or(50),
472            offset: self.offset.unwrap_or(0),
473            ..Default::default()
474        };
475
476        // Parse tags (comma-separated, prefix with - for exclude)
477        if let Some(tags_str) = self.tags {
478            query.tags = tags_str
479                .split(',')
480                .map(|t| {
481                    let tag = t.trim();
482                    if let Some(tag_name) = tag.strip_prefix('-') {
483                        TagFilter {
484                            tag: tag_name.to_string(),
485                            mode: TagFilterMode::Exclude,
486                        }
487                    } else {
488                        TagFilter {
489                            tag: tag.to_string(),
490                            mode: TagFilterMode::Include,
491                        }
492                    }
493                })
494                .collect();
495        }
496
497        // Parse type/kind filter
498        if let Some(kind_str) = self.kind {
499            query.kinds = kind_str
500                .split(',')
501                .filter_map(|k| match k.trim() {
502                    "Image" | "image" => Some(hippo_core::MemoryKind::Image {
503                        width: 0,
504                        height: 0,
505                        format: String::new(),
506                    }),
507                    "Video" | "video" => Some(hippo_core::MemoryKind::Video {
508                        duration_ms: 0,
509                        format: String::new(),
510                    }),
511                    "Audio" | "audio" => Some(hippo_core::MemoryKind::Audio {
512                        duration_ms: 0,
513                        format: String::new(),
514                    }),
515                    "Document" | "document" => Some(hippo_core::MemoryKind::Document {
516                        format: hippo_core::DocumentFormat::Other(String::new()),
517                        page_count: None,
518                    }),
519                    "Code" | "code" => Some(hippo_core::MemoryKind::Code {
520                        language: String::new(),
521                        lines: 0,
522                    }),
523                    _ => None,
524                })
525                .collect();
526        }
527
528        // Parse sort order
529        if let Some(sort_str) = self.sort {
530            query.sort = match sort_str.as_str() {
531                "Relevance" => hippo_core::SortOrder::Relevance,
532                "DateNewest" => hippo_core::SortOrder::DateNewest,
533                "DateOldest" => hippo_core::SortOrder::DateOldest,
534                "NameAsc" => hippo_core::SortOrder::NameAsc,
535                "NameDesc" => hippo_core::SortOrder::NameDesc,
536                "SizeAsc" => hippo_core::SortOrder::SizeAsc,
537                "SizeDesc" => hippo_core::SortOrder::SizeDesc,
538                _ => {
539                    return Err(ApiError::BadRequest(format!(
540                        "Invalid sort order: {}",
541                        sort_str
542                    )))
543                }
544            };
545        }
546
547        Ok(query)
548    }
549}

576// Error Handling

578
579#[derive(Debug)]
580enum ApiError {
581    NotFound(String),
582    BadRequest(String),
583    InternalError(String),
584    HippoError(hippo_core::HippoError),
585}
586
587impl From<hippo_core::HippoError> for ApiError {
588    fn from(err: hippo_core::HippoError) -> Self {
589        ApiError::HippoError(err)
590    }
591}
592
593impl IntoResponse for ApiError {
594    fn into_response(self) -> Response {
595        let (status, message) = match self {
596            ApiError::NotFound(msg) => (StatusCode::NOT_FOUND, msg),
597            ApiError::BadRequest(msg) => (StatusCode::BAD_REQUEST, msg),
598            ApiError::InternalError(msg) => (StatusCode::INTERNAL_SERVER_ERROR, msg),
599            ApiError::HippoError(err) => {
600                error!("Hippo error: {:?}", err);
601                (
602                    StatusCode::INTERNAL_SERVER_ERROR,
603                    format!("Internal error: {}", err),
604                )
605            }
606        };
607
608        let body = Json(serde_json::json!({
609            "error": message
610        }));
611
612        (status, body).into_response()
613    }
614}
